\section{Method}
\label{sec:method}

Let $P_\tau$ denote the observed transaction price at tick time $\tau$. The observed price can be decomposed as shown in Equation \ref{eq:price_decomp}:
\begin{equation}
    P_\tau = P_\tau^*+\eta_\tau
    \label{eq:price_decomp}
\end{equation}
where $P_\tau^*$ is the latent efficient price and $\eta_\tau$ represents microstructure noise. The pre-averaged price is defined in Equation \ref{eq:pre_avg} as
\begin{equation}
    \bar P_t = \frac{1}{k} \sum_{j=0}^{k-1} P_{t-j}
    \label{eq:pre_avg}
\end{equation}
where $t$ now indexes the last tick in each averaging block. Returns are then constructed as shown in Equation \ref{eq:returns}:
\begin{equation}
    r_t=\bar P_t - \bar P_{t-1}
    \label{eq:returns}
\end{equation}

\subsection{Time dependent models}

AR(p)
\begin{equation}
    Y_t = c + \phi_1Y_{t-1} + \phi_2Y_{t-2} + \dots + \phi_pY_{t-p} + \epsilon_t
\end{equation}

MA(q)
\begin{equation}
    Y_t = \mu + \epsilon_t + \theta_1\epsilon_{t-1} + \theta_2\epsilon_{t-2} + \dots + \theta_q\epsilon_{t-q}
\end{equation}

ARMA(p,q)
\begin{equation}
    Y_t=c+\sum_{i=1}^p\phi_iY_{t-1}+\epsilon_t+\sum_{j=1}^q\theta_j\epsilon_{t-j}
\end{equation}

ARIMAX(p,d,q)
\begin{equation}
    Y_t^{'} = c + \beta Xt +  \sum_{i=1}^p\phi_iY^{'}_{t-1}+\epsilon_t+\sum_{j=1}^q\theta_j\epsilon_{t-j}
\end{equation}

ARCH(q)
\begin{equation}
    \sigma_t^2 = \alpha_0+\alpha_1\epsilon_{t-1}^2+\dots+\alpha_q\epsilon_{t-q}^2
\end{equation}

GARCH(p,q)
\begin{equation}
    \sigma_t^2 = \alpha_0 + \sum_{i=1}^q\alpha_i\epsilon_{t-i}^2+\sum_{j=1}^p
\beta_j\sigma_{t-j}^2
\end{equation}

As a benchmark, we consider a single-regime auto-regressive model of order one, given by Equation \ref{eq:baseline_ar}:
\begin{equation}
    r_t=\mu+\phi r_{t-1} + \varepsilon_t
    \label{eq:baseline_ar}
\end{equation}
where $\mu$ is the unconditional mean return, $\phi$ measures linear serial dependence and $\varepsilon_t$ is an innovative term with $\varepsilon_t\sim \mathcal{N}(0,\sigma^2)$.

\subsection{Hidden Markov Model with Regime-Switching Auto-regressive Dynamics}

Let $S_t \in \{ 1,\dots,K \}$ denote an unobserved discrete-time Markov chain representing the dependence regime at time $t$. The transition probabilities are defined in Equation \ref{eq:trans_probs}:
\begin{equation}
    \mathbb{P}(S_t=j \mid S_{t-1} = i) = p_{ij}
    \label{eq:trans_probs}
\end{equation}
with $p_{ij} \geq 0$ and $\sum_{j=1}^K p_{ij} = 1$ for all $i$.

Conditional on the regime $S_t=k$, returns follow an auto-regressive process described by Equation \ref{eq:regime_ar}:
\begin{equation}
    r_t = \mu_k + \phi_kr_{t-1} + \varepsilon_t, \quad \varepsilon_t \sim \mathcal{N}(0,\sigma_k^2)
    \label{eq:regime_ar}
\end{equation}

Here $\mu_k$ is the regime-specific mean, $\phi_k$ captures the strength of the short-term dependence in regime $k$ and $\sigma_k^2$ is the regime-specific innovation variance.

Regimes with $\phi_k$ close to zero are interpreted as noise-dominated periods, while regimes with larger positive $\phi_k$ correspond to short-term momentum or price returns.

\subsection{Modeling volatility}

Allowing $\sigma_k^2$ to vary across regimes captures abrupt changes in market uncertainty. As an extension, conditional heteroskedasticity can be introduced via a regime-dependent ARCH specification, shown in Equation \ref{eq:arch}:
\begin{equation}
    \sigma_{k,t}^2 = \omega_k + \alpha_k\varepsilon_{t-1}^2
    \label{eq:arch}
\end{equation}
where $\omega_k\geq0$ and $\alpha_k\geq0$. Building on this, volatility persistence can be incorporated through a regime-dependent GARCH specification, see Equation \ref{eq:garch}:
\begin{equation}
    \sigma_{k,t}^2 = \omega_k + \alpha_k\varepsilon_{t-1}^2 + \beta_k\sigma_{k,t-1}^2,
    \label{eq:garch}
\end{equation}
where $\beta_k\geq0$. This extension allows volatility shocks to decay gradually over time within each regime rather than affecting only the next period.

\subsection{Complete parameter set and emission density}

The complete set of HMM parameters is
\begin{equation}
    \theta = \bigl\{\pi,\,\mathbf{P},\,\mu_k,\,\phi_k,\,\sigma_k^2 : k = 1,\dots,K\bigr\},
    \label{eq:params}
\end{equation}
where $\pi = (\pi_1,\dots,\pi_K)^\top$ with $\pi_k = \mathbb{P}(S_1 = k)$ is the initial state distribution, $\mathbf{P} = (p_{ij})$ is the $K\times K$ transition matrix, and $\mu_k$, $\phi_k$, $\sigma_k^2$ are the regime-specific AR parameters introduced in Equation \ref{eq:regime_ar}. Conditional on $S_t = k$ and the previous return $r_{t-1}$, the emission density is
\begin{equation}
    f_k(r_t \mid r_{t-1};\theta)
    = \frac{1}{\sqrt{2\pi\sigma_k^2}}
    \exp\!\left(-\frac{(r_t - \mu_k - \phi_k r_{t-1})^2}{2\sigma_k^2}\right).
    \label{eq:emission}
\end{equation}

\subsection{Likelihood and the forward algorithm}

Direct maximisation of the observed-data log-likelihood $\ell(\theta) = \log \mathbb{P}(r_{1:T};\theta)$ is intractable because it requires summing over all $K^T$ possible state sequences. The \emph{forward algorithm} \parencite{rabiner1989,cappe2005} computes the likelihood recursively. Define the forward variable
\begin{equation}
    \alpha_t(k) = \mathbb{P}(r_1,\dots,r_t,\,S_t = k \mid \theta),
    \label{eq:forward}
\end{equation}
initialised as $\alpha_1(k) = \pi_k f_k(r_1)$ and propagated via
\begin{equation}
    \alpha_t(k) = \left[\sum_{i=1}^{K} \alpha_{t-1}(i)\,p_{ik}\right] f_k(r_t \mid r_{t-1};\theta),
    \quad t = 2,\dots,T.
    \label{eq:forward_rec}
\end{equation}
The observed-data likelihood then equals $\mathbb{P}(r_{1:T};\theta) = \sum_{k=1}^{K}\alpha_T(k)$.

\subsection{Viterbi decoding}

Given estimated parameters $\hat\theta$, the most probable latent state sequence is
\begin{equation}
    \hat S_{1:T} = \operatorname*{arg\,max}_{S_{1:T}} \mathbb{P}(S_{1:T} \mid r_{1:T};\hat\theta).
    \label{eq:viterbi_obj}
\end{equation}
The \emph{Viterbi algorithm} \parencite{rabiner1989} solves this by dynamic programming. Define
\begin{equation}
    \delta_t(k) = \max_{S_{1:t-1}} \mathbb{P}(S_{1:t-1},\, S_t = k,\, r_{1:t} \mid \hat\theta),
    \label{eq:viterbi}
\end{equation}
initialised as $\delta_1(k) = \pi_k f_k(r_1)$ and updated via
\begin{equation}
    \delta_t(k) = \max_{1 \leq i \leq K}\left[\delta_{t-1}(i)\,p_{ik}\right] f_k(r_t \mid r_{t-1};\hat\theta).
    \label{eq:viterbi_rec}
\end{equation}
Storing the maximising state at each step, the optimal path $\hat S_{1:T}$ is recovered by backtracking from $\hat S_T = \operatorname*{arg\,max}_k \delta_T(k)$.

\subsection{Parameter estimation via Baum--Welch}

Parameters are estimated by maximum likelihood using the \emph{Baum--Welch algorithm} \parencite{cappe2005}, an instance of the Expectation--Maximisation (EM) procedure of \textcite{dempster1977}. Alongside the forward variable $\alpha_t(k)$, define the backward variable
\begin{equation}
    \beta_t(k) = \mathbb{P}(r_{t+1},\dots,r_T \mid S_t = k,\,\theta),
    \label{eq:backward}
\end{equation}
initialised as $\beta_T(k)=1$ and propagated as $\beta_t(k) = \sum_{j=1}^K p_{kj}\,f_j(r_{t+1}\mid r_t;\theta)\,\beta_{t+1}(j)$. The E-step computes the smoothed state occupancy probabilities
\begin{equation}
    \gamma_t(k) = \mathbb{P}(S_t = k \mid r_{1:T};\theta)
    = \frac{\alpha_t(k)\,\beta_t(k)}{\displaystyle\sum_{j=1}^{K}\alpha_t(j)\,\beta_t(j)}
    \label{eq:gamma}
\end{equation}
and the pairwise transition probabilities
\begin{equation}
    \xi_t(i,j) = \mathbb{P}(S_{t-1}=i,\,S_t=j \mid r_{1:T};\theta)
    \;\propto\; \alpha_{t-1}(i)\,p_{ij}\,f_j(r_t \mid r_{t-1};\theta)\,\beta_t(j).
    \label{eq:xi}
\end{equation}
The M-step updates the parameters in closed form. Transition probabilities are updated as
\begin{equation}
    p_{ij}^{\text{new}} = \frac{\displaystyle\sum_{t=2}^{T}\xi_t(i,j)}{\displaystyle\sum_{t=2}^{T}\gamma_{t-1}(i)},
    \label{eq:mstep_p}
\end{equation}
and the regime-specific AR parameters $(\mu_k,\phi_k,\sigma_k^2)$ are updated via weighted least squares with $\gamma_t(k)$ as observation weights. The E- and M-steps are iterated to convergence of the log-likelihood.

\subsection{Model selection}

The number of regimes $K$ is treated as a hyperparameter. For each candidate $K$ we compute the Bayesian Information Criterion
\begin{equation}
    \mathrm{BIC}(K) = -2\,\ell(\hat\theta_K) + p_K \log T,
    \label{eq:bic}
\end{equation}
where $p_K = K^2 + 3K - 1$ counts the free parameters (transition matrix entries, regime-specific AR intercepts, slopes, and variances, and initial distribution) and $T$ is the number of observations. The preferred model minimises $\mathrm{BIC}(K)$. The stronger complexity penalty of BIC relative to AIC is appropriate given the large sample sizes typical of tick data. Empirical evidence from HMM-based intraday trading strategies suggests that a small number of regimes, typically $K \in \{2,3\}$, is sufficient to capture the dominant market dynamics \parencite{christensen2020}, providing prior support for evaluating candidate models in this range.

\subsection{Financial metrics}

To evaluate the performance of the trading strategy, a set of financial metrics is computed. These metrics capture execution quality, trade-level performance, and portfolio-level risk and return characteristics.

\subsubsection{Execution metrics}

Average slippage measures the average absolute difference between the executed price and the intended price.

\begin{equation}
S_{\text{avg}}
=
\frac{1}{N}
\sum_{i=1}^{N}
\left|
P_{\text{exec},i}
-
P_{\text{int},i}
\right|
\end{equation}

Fill rate measures the fraction of the requested trading volume that was actually executed.

\begin{equation}
\text{Fill Rate}
=
\frac{\sum_{i=1}^{N} V_{\text{exec},i}}
{\sum_{i=1}^{N} V_{\text{req},i}}
\end{equation}

\subsubsection{Trade-level metrics}

The average holding time measures the average time a position remains open.

\begin{equation}
\overline{T}
=
\frac{1}{N}
\sum_{i=1}^{N}
\left(
t_{\text{close},i}
-
t_{\text{open},i}
\right)
\end{equation}

The win rate measures the fraction of trades that are profitable.

\begin{equation}
W
=
\frac{N_{\text{win}}}{N_{\text{total}}}
\end{equation}

Profit factor measures the ratio of total profits to total losses.

\begin{equation}
\text{PF}
=
\frac{\sum \text{Gross Profits}}
{\sum |\text{Gross Losses}|}
\end{equation}

Average net profit per trade after transaction costs is defined as

\begin{equation}
\text{PnL}_{\text{avg}}
=
\frac{1}{N}
\sum_{i=1}^{N}
\left(
(P_{\text{exit},i}-P_{\text{entry},i})Q_i
-
c_i
\right)
\end{equation}

Cumulative net profit and loss represents the total economic value of the strategy rather than just per-trade averages.

\begin{equation}
\text{PnL}_{\text{tot}}
=
\sum_{i=1}^{N}
\left(
(P_{\text{exit},i}-P_{\text{entry},i})Q_i
-
c_i
\right)
\end{equation}

Average trade return as a fraction of capital committed is given by

\begin{equation}
\bar r_{\text{trade}}
=
\frac{1}{N}
\sum_{i=1}^{N}
\frac{
(P_{\text{exit},i}-P_{\text{entry},i})Q_i - c_i
}{
P_{\text{entry},i}Q_i
}
\end{equation}

The average gain and average loss separate the magnitude of winning and losing trades.

\begin{equation}
\overline{\text{Gain}}
=
\frac{1}{N_{\text{win}}}
\sum_{i \in \mathcal{W}}
\text{PnL}_i
\end{equation}

\begin{equation}
\overline{\text{Loss}}
=
\frac{1}{N_{\text{loss}}}
\sum_{i \in \mathcal{L}}
\left|\text{PnL}_i\right|
\end{equation}

Expectancy combines win rate and win/loss magnitude to measure expected profit per trade.

\begin{equation}
\text{Expectancy}
=
W \cdot \overline{\text{Gain}}
-
(1-W)\cdot \overline{\text{Loss}}
\end{equation}

\subsubsection{Risk and return metrics}

The maximum drawdown measures the largest peak-to-trough decline in portfolio value over the period.

\begin{equation}
\text{MDD}
=
\max_{\tau \in (0,T)}
\left(
\max_{t \in (0,\tau)}
\frac{V(t)-V(\tau)}{V(t)}
\right)
\end{equation}

The Sharpe ratio measures excess return per unit of total volatility.

\begin{equation}
\text{Sharpe}
=
\frac{\mathbb{E}[R_p-R_f]}{\sigma_p}
\end{equation}

The Sortino ratio measures the average excess return per unit of downside volatility.

\begin{equation}
\text{Sortino}
=
\frac{\mathbb{E}[R_p-R_f]}{\sigma_d}
\end{equation}

Return volatility reports the variability of returns directly.

\begin{equation}
\sigma_R
=
\sqrt{
\frac{1}{T-1}
\sum_{t=1}^{T}
(R_t-\bar R)^2
}
\end{equation}

where

\begin{equation}
\bar R
=
\frac{1}{T}
\sum_{t=1}^{T} R_t
\end{equation}

The Calmar ratio relates returns to drawdown risk.

\begin{equation}
\text{Calmar}
=
\frac{\mathbb{E}[R_p-R_f]}{\text{MDD}}
\end{equation}

Alternatively, it may be defined using annualized return:

\begin{equation}
\text{Calmar}
=
\frac{R_{\text{ann}}}{\text{MDD}}
\end{equation}

\subsubsection{Other metrics}

Turnover measures how intensively the strategy trades relative to available capital, where $A_0$ is the initial capital.

\begin{equation}
\text{Turnover}
=
\frac{
\sum_{i=1}^{N}
|Q_i|P_{\text{entry},i}
}{
A_0
}
\end{equation}

Exposure time measures the fraction of total sample time during which the strategy has an open position.

\begin{equation}
\text{Exposure}
=
\frac{
\sum_{i=1}^{N}
\left(
t_{\text{close},i}
-
t_{\text{open},i}
\right)
}{
T_{\text{sample}}
}
\end{equation}
